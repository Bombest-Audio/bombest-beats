<objective>
Apply the two surgical code changes that unblock reliable Espresso CI runs: (1) add the KVM
permissions step to `pre-merge.yml` so the x86_64 emulator uses hardware acceleration instead
of pure software emulation, and (2) remove the `assumeTrue(false)` skip path from
`AndroidAutoBrowseTest` so service bind failures produce hard test failures rather than
silent skips. Also add the discretionary improvements: job timeout cap and pinned
emulator-runner version.

Purpose: The missing KVM permissions step is the confirmed root cause of all 10 CI failures on
PR #36. Without KVM, `disable-linux-hw-accel: auto` sets `disable Linux hardware acceleration:
true`, causing either "No compatible devices connected" (emulator boots but Gradle rejects it)
or GC-induced UI timeouts (Compose never renders login form within 60s). Fixing KVM fixes both
failure modes. Removing `assumeTrue` satisfies D-03.

Output: Updated `pre-merge.yml` with KVM step + timeout + pinned version, and updated
`AndroidAutoBrowseTest.kt` with hard-failure path and try/finally browser.release().
</objective>

The `browser` variable is assigned after the `try/catch` block above. Restructure the test
body so `browser.release()` is in a `finally` block. Declare `var browser: MediaBrowser? = null`
before the try block, then reassign inside, and release in finally:

```kotlin
        var browser: MediaBrowser? = null
        try {
            browser = browserFuture.get(60, TimeUnit.SECONDS)
        } catch (e: TimeoutException) {
            browserFuture.cancel(true)
            throw AssertionError("BombestMediaService did not bind within 60 s on this runner")
        }
        assertNotNull("MediaBrowser must bind to BombestMediaService", browser)

        try {
            // Fetch root, then its children — Auto does exactly this handshake.
            val rootLatch = CountDownLatch(1)
            var rootId: String? = null
            InstrumentationRegistry.getInstrumentation().runOnMainSync {
                val rootFuture = browser!!.getLibraryRoot(null)
                rootFuture.addListener({
                    rootId = rootFuture.get().value?.mediaId
                    rootLatch.countDown()
                }, MoreExecutors.directExecutor())
            }
            assertTrue("root resolved", rootLatch.await(30, TimeUnit.SECONDS))
            assertNotNull("root mediaId must be non-null", rootId)

            val childrenLatch = CountDownLatch(1)
            var children: List<MediaItem> = emptyList()
            InstrumentationRegistry.getInstrumentation().runOnMainSync {
                val childrenFuture = browser!!.getChildren(rootId!!, 0, 50, null)
                childrenFuture.addListener({
                    children = childrenFuture.get().value.orEmpty().toList()
                    childrenLatch.countDown()
                }, MoreExecutors.directExecutor())
            }
            assertTrue("children resolved", childrenLatch.await(30, TimeUnit.SECONDS))

            // The four top-level categories Auto users expect to see.
            val childIds = children.map { it.mediaId }.toSet()
            assertTrue("browse root should include Songs, got: $childIds", childIds.contains("songs"))
            assertTrue(
                "browse root should include Playlists, got: $childIds",
                childIds.contains("playlists"),
            )
            assertTrue(
                "browse root should include Recent, got: $childIds",
                childIds.contains("recent"),
            )
        } finally {
            InstrumentationRegistry.getInstrumentation().runOnMainSync {
                browser?.release()
            }
        }
    }
```
